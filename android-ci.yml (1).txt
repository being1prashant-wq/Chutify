name: ModihMusic Android CI

on:
  push:
    branches:
      - main
      - master
  pull_request:
    branches:
      - main
      - master
  workflow_dispatch:

permissions:
  contents: read

jobs:
  build:
    name: Build ModihMusic
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "17"

      - name: Find ZIP file
        shell: bash
        run: |
          set -e

          ZIP_FILE=""

          if [ -f "Moody_Source_Fixed.zip" ]; then
            ZIP_FILE="Moody_Source_Fixed.zip"
          elif [ -f "Moody.zip" ]; then
            ZIP_FILE="Moody.zip"
          else
            ZIP_FILE="$(find . -maxdepth 2 -type f -name '*.zip' | head -n 1 || true)"
          fi

          if [ -z "$ZIP_FILE" ]; then
            echo "ERROR: No ZIP file found in repository root."
            find . -maxdepth 3 -type f -print
            exit 1
          fi

          echo "ZIP_FILE=$ZIP_FILE" >> "$GITHUB_ENV"
          echo "Using ZIP: $ZIP_FILE"

      - name: Extract Android project
        shell: bash
        run: |
          set -e

          rm -rf android-project
          mkdir -p android-project

          unzip -q "$ZIP_FILE" -d android-project

          echo "Extracted ZIP contents:"
          find android-project -maxdepth 4 -type f | sort

      - name: Locate Android project
        shell: bash
        run: |
          set -e

          SETTINGS_FILE="$(find android-project -type f \( -name 'settings.gradle.kts' -o -name 'settings.gradle' \) -print -quit)"

          if [ -z "$SETTINGS_FILE" ]; then
            echo "ERROR: settings.gradle(.kts) not found after extraction."
            echo
            echo "Extracted files:"
            find android-project -maxdepth 8 -type f | sort
            exit 1
          fi

          PROJECT_DIR="$(dirname "$SETTINGS_FILE")"
          echo "PROJECT_DIR=$PROJECT_DIR" >> "$GITHUB_ENV"
          echo "Android project found at: $PROJECT_DIR"

      # --- NEW: the delivered zip has every wrapper file except the jar
      # itself (a compiled binary that couldn't be produced without a
      # network connection or a local Gradle install). setup-gradle
      # guarantees `gradle` is callable regardless of what's already on
      # this runner image, then it's used once to generate the real jar.
      - name: Setup Gradle
        uses: gradle/actions/setup-gradle@v6

      - name: Generate Gradle wrapper jar if missing
        shell: bash
        run: |
          set -e
          cd "$PROJECT_DIR"
          if [ ! -f "gradle/wrapper/gradle-wrapper.jar" ]; then
            echo "gradle-wrapper.jar not present — generating it."
            gradle wrapper --gradle-version 9.3.0
          else
            echo "gradle-wrapper.jar already present, nothing to do."
          fi

      - name: Verify Android project
        shell: bash
        run: |
          set -e
          cd "$PROJECT_DIR"

          test -f "gradlew"
          test -f "gradle/wrapper/gradle-wrapper.properties"
          test -f "gradle/wrapper/gradle-wrapper.jar"
          test -d "app"

          echo "Project structure OK."

      - name: Make Gradle wrapper executable
        shell: bash
        run: |
          cd "$PROJECT_DIR"
          chmod +x ./gradlew

      - name: Show Gradle version
        shell: bash
        run: |
          cd "$PROJECT_DIR"
          ./gradlew --version

      - name: Clean project
        shell: bash
        env:
          GRADLE_OPTS: "-Xmx4g -Dfile.encoding=UTF-8"
        run: |
          cd "$PROJECT_DIR"
          ./gradlew clean --no-daemon --no-parallel --stacktrace --console=plain

      - name: Build Debug APK
        shell: bash
        env:
          YOUTUBE_API_KEY: ${{ secrets.YOUTUBE_API_KEY }}
          GRADLE_OPTS: "-Xmx4g -Dfile.encoding=UTF-8"
        run: |
          set -o pipefail
          cd "$PROJECT_DIR"
          ./gradlew :app:assembleDebug \
            --no-daemon \
            --no-parallel \
            --stacktrace \
            --console=plain \
            2>&1 | tee "$GITHUB_WORKSPACE/build.log"

      - name: Show build error
        if: failure()
        shell: bash
        run: |
          echo ""
          echo "======================================"
          echo "          BUILD FAILED"
          echo "======================================"
          echo ""

          if [ -f "$GITHUB_WORKSPACE/build.log" ]; then
            grep -n -E "e: |error:|ERROR:|Execution failed|FAILED|Caused by:" "$GITHUB_WORKSPACE/build.log" | head -n 200 || true
            echo ""
            tail -n 250 "$GITHUB_WORKSPACE/build.log"
          fi

      - name: Verify APK
        if: success()
        shell: bash
        run: |
          set -e
          APK="android-project/**/app/build/outputs/apk/debug/app-debug.apk"

          found="$(find android-project -type f -path '*/app/build/outputs/apk/debug/app-debug.apk' | head -n 1 || true)"

          if [ -z "$found" ]; then
            echo "ERROR: APK not generated."
            find android-project -type f -name "*.apk" -print || true
            exit 1
          fi

          echo "APK generated:"
          ls -lh "$found"

      - name: Upload Debug APK
        if: success()
        uses: actions/upload-artifact@v4
        with:
          name: ModihMusic-debug-apk
          path: android-project/**/app/build/outputs/apk/debug/*.apk
          if-no-files-found: error

      - name: Upload build log
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: ModihMusic-build-log
          path: build.log
          if-no-files-found: ignore
